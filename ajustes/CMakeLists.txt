cmake_minimum_required(VERSION 3.20)
project(vitrum-ajustes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Gui Widgets Quick Qml DBus)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6KCMUtils REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_executable(vitrum-ajustes src/main.cpp src/kcmview.cpp)
target_link_libraries(vitrum-ajustes Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::DBus KF6::WindowSystem KF6::ConfigCore KF6::Service KF6::CoreAddons KF6::KCMUtilsQuick)
install(TARGETS vitrum-ajustes DESTINATION bin)
